socket.on('metric_update', function(payload) {
    if (computeChart.data.labels.length > 20) {
        computeChart.data.labels.shift();
        computeChart.data.datasets[0].data.shift();
    }
    computeChart.data.labels.push(payload.time);
    computeChart.data.datasets[0].data.push(payload.value);
    computeChart.update('none'); // Update without animation for performance
});
3. Backend: Real-Time Scraper (src/reaper/collectors/azure_collector.py)Instead of just fetching prices, we now fetch Metrics.Pythonfrom azure.mgmt.monitor import MonitorManagementClient

def get_vm_metrics(resource_id):
    client = MonitorManagementClient(credential, subscription_id)
    metrics = client.metrics.list(
        resource_id,
        timespan='PT1H',
        interval='PT1M',
        metricnames='Percentage CPU',
        aggregation='Average'
    )
    # Extract the most recent value
    latest_val = metrics.value[0].timeseries[0].data[-1].average
    return latest_val

5. Performance Optimization (DevOps Tip)
Running real-time charts can be heavy on the browser. Since you are using Ubuntu 24.04 and gevent, ensure you set a throttle on your socket emissions. Don't push metrics faster than once every 5–10 seconds for Azure services, as the Azure Monitor API itself doesn't update faster than that for standard metrics.
