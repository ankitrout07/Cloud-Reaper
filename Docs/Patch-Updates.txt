By extracting live performance metrics (vCPU utilization, memory footprints, and network throughput) from Prometheus via its PromQL HTTP API, your Python intelligence layer can cross-reference actual resource usage against the raw cost metrics stored in your PostgreSQL region_price_cache table. This bridges the gap between raw capacity and actual deployment waste.Here is the technical architecture, implementation engine, and validation method to inject this capability directly into Cloud-Reaper.📐 The Theory: Mapping Telemetry to Financial WasteCommercial tools often look at cloud bills statically. To calculate true financial waste dynamically, we cross-reference pricing metrics with real-time utilization data over a time window ($T$):$$\text{Waste Rate (\%)} = 100 \times \left( 1.0 - \frac{\frac{1}{T}\int_{0}^{T} \text{Metric}_{\text{actual}}(t) \, dt}{\text{Metric}_{\text{provisioned}}} \right)$$If an instance runs at an average of 8% CPU utilization over a 7-day window, your engine flags the exact dollar volume being wasted per hour on idle headroom.🛠️ Implementation EngineThis production-grade core service connects directly to your internal Prometheus data layer, executes real-time metric evaluations, and surfaces precise cost-optimization recommendations.1. Prometheus Metric Collector (src/reaper/collectors/prometheus_finops.py)Python# src/reaper/collectors/prometheus_finops.py
import requests
import time
import logging

logger = logging.getLogger("reaper.collector.prometheus")

class PrometheusFinOpsCollector:
    def __init__(self, prometheus_url: str = "http://localhost:9090"):
        self.api_endpoint = f"{prometheus_url}/api/v1/query"

    def fetch_node_avg_cpu_utilization(self, duration_window: str = "7d") -> dict:
        """Executes PromQL queries to calculate average CPU utilization across clusters."""
        # PromQL query tracking average non-idle CPU percentage across the fleet
        promql_query = f"100 - (avg by (instance) (rate(node_cpu_seconds_total{{mode='idle'}}[{duration_window}])) * 100)"
        
        try:
            response = requests.get(self.api_endpoint, params={'query': promql_query}, timeout=10)
            if response.status_code != 200:
                logger.error(f"Prometheus query failure. HTTP Status: {response.status_code}")
                return {}
            
            results = response.json().get("data", {}).get("result", [])
            utilization_map = {}
            for item in results:
                instance = item.get("metric", {}).get("instance", "unknown")
                # Extract the final scalar metric reading tuple [timestamp, value]
                value = float(item.get("value", [0, 0])[1])
                utilization_map[instance] = round(value, 2)
            
            return utilization_map
        except requests.exceptions.RequestException as e:
            logger.critical(f"Failed to reach Prometheus API backend: {str(e)}")
            return {}
2. FinOps Metric Cross-Reference Engine (src/reaper/engine/metrics_analyzer.py)This service maps your live Prometheus performance readings directly to your cloud inventory costs to calculate exact financial waste.Python# src/reaper/engine/metrics_analyzer.py
from reaper.collectors.prometheus_finops import PrometheusFinOpsCollector

class FinOpsTelemetryAnalyzer:
    def __init__(self, prometheus_url: str):
        self.prom_collector = PrometheusFinOpsCollector(prometheus_url)

    def analyze_compute_waste_index(self, active_inventory: list) -> list:
        """Cross-references Prometheus utilization data against actual asset cost metrics."""
        cpu_metrics = self.prom_collector.fetch_node_avg_cpu_utilization()
        optimization_insights = []

        for asset in active_inventory:
            instance_ip = asset.get("private_ip")
            monthly_cost = asset.get("monthly_cost", 0.0)
            
            # Match inventory assets with active Prometheus telemetry instances
            matching_instance = next((inst for inst in cpu_metrics if instance_ip in inst), None)
            avg_utilization = cpu_metrics.get(matching_instance, 50.0) if matching_instance else 50.0

            # Underutilization threshold rule: Flag instances running under 10% capacity
            if avg_utilization < 10.0:
                potential_savings = monthly_cost * 0.50 # Target a baseline 50% scale-down downsize
                insight = {
                    "resource_id": asset.get("resource_id"),
                    "current_sku": asset.get("sku_size"),
                    "avg_cpu_utilization": f"{avg_utilization}%",
                    "monthly_waste_impact": f"${potential_savings:.2f}",
                    "remediation_action": "DOWNGRADE_SKU_FAMILY"
                }
                optimization_insights.append(insight)

        return optimization_insights

1. Raw Telemetry Extraction (From Prometheus HTTP API)
Prometheus exposes a native REST API under the /api/v1 namespace that outputs raw website performance metric payloads completely structured in JSON.

The Target Endpoint Format
To pull real-time historical usage data of your website (e.g., HTTP request counts or duration rates over a range), your engine-go or Python collector queries the Prometheus Range Query Engine:

Plaintext
GET http://<prometheus-host>:9090/api/v1/query_range?query=rate(http_requests_total[5m])&start=2026-05-20T12:00:00Z&end=2026-05-20T17:00:00Z&step=15s
The Inbound JSON Payload Structure
Prometheus returns an envelope structure containing a timestamp/value matrix:

JSON
{
  "status": "success",
  "data": {
    "resultType": "matrix",
    "result": [
      {
        "metric": {
          "__name__": "http_requests_total",
          "instance": "production-web-server-01",
          "job": "website-frontend",
          "status_code": "200"
        },
        "values": [
          [1779285600, "42.5"], 
          [1779285615, "48.1"]  
        ]
      }
    ]
  }
}
How Cloud-Reaper Leverages This
Your backend parses the values array (Format: [Unix_Timestamp, "Value_String"]).

It calculates the workload utilization trends and matches the instance label against your PostgreSQL resources table.

If the request volume is near zero for days, Cloud-Reaper flags it as a Zombie Resource to save costs.

2. High-Performance Client-Side Parsing Engine
Add this script to the bottom of your template file or your global static asset JS file. It hooks directly into the browser's native HTML5 Drag and Drop API events (dragover, dragleave, drop), processes the JSON payload using the asynchronous FileReader API, and dynamically extracts and updates your dashboard statistics.

JavaScript
// Append this implementation script logic to your dashboard web layouts
document.addEventListener("DOMContentLoaded", () => {
    const dropZone = document.getElementById("drop-zone");
    const fileInput = document.getElementById("file-input");
    const dropZoneText = document.getElementById("drop-zone-text");

    // Click handler to route drop-zone interaction to the hidden native file picker
    dropZone.addEventListener("click", () => fileInput.click());

    // Prevent default browser behaviors for file dropping
    ["dragenter", "dragover", "dragleave", "drop"].forEach(eventName => {
        window.addEventListener(eventName, (e) => e.preventDefault(), false);
    });

    // Visual hover states using Cloud-Reaper cyan border styling highlights
    dropZone.addEventListener("dragover", () => {
        dropZone.style.border = "2px dashed #00f3ff";
        dropZone.style.background = "rgba(0, 243, 255, 0.05)";
    });

    ["dragleave", "drop"].forEach(eventName => {
        dropZone.addEventListener(eventName, () => {
            dropZone.style.border = "2px dashed rgba(0, 243, 255, 0.3)";
            dropZone.style.background = "rgba(0, 0, 0, 0.2)";
        });
    });

    // Handle dropping execution hook
    dropZone.addEventListener("drop", (e) => {
        const files = e.dataTransfer.files;
        if (files.length > 0) {
            processTelemetryJsonFile(files[0]);
        }
    });

    // Handle manual browse selection change hook
    fileInput.addEventListener("change", (e) => {
        if (e.target.files.length > 0) {
            processTelemetryJsonFile(e.target.files[0]);
        }
    });

    function processTelemetryJsonFile(file) {
        if (file.type !== "application/json" && !file.name.endsWith(".json")) {
            alert("CRITICAL ERROR: Cloud-Reaper ingestion only accepts valid .json telemetry reports.");
            return;
        }

        const reader = new FileReader();
        reader.onload = function(event) {
            try {
                const parsedData = JSON.parse(event.target.result);
                
                // Update UI text to demonstrate successful file parsing action
                dropZoneText.innerHTML = `<i class="fas fa-check-circle" style="color:#00ff66;"></i> Successfully Loaded: <strong>${file.name}</strong>`;
                
                // Execute standard metric processing mapping over Prometheus metric models
                renderExtractedStats(parsedData);
            } catch (err) {
                alert("PARSING ERROR: Provided file is not a structurally sound JSON payload document.");
            }
        };
        reader.readAsText(file);
    }

    function renderExtractedStats(jsonData) {
        // Safe, basic heuristic parsing strategy looking for native Prometheus data structures
        const resultsArray = jsonData?.data?.result || [];
        const targetsCount = resultsArray.length || 1;
        
        // Simulating analytical cost evaluations directly from payload metadata readings
        let totalCalculatedCpu = 0;
        resultsArray.forEach(item => {
            const lastValue = item.values ? parseFloat(item.values[item.values.length - 1][1]) : parseFloat(item.value?.[1] || 15);
            totalCalculatedCpu += isNaN(lastValue) ? 15 : lastValue;
        });

        const avgCpu = (totalCalculatedCpu / targetsCount).toFixed(1);
        const wasteFactor = avgCpu < 15 ? (100 - parseFloat(avgCpu)).toFixed(0) : 0;

        // Populate visual stats numbers into the DOM containers instantly
        document.getElementById("stat-targets").innerText = targetsCount;
        document.getElementById("stat-cpu").innerText = `${avgCpu}%`;
        document.getElementById("stat-waste").innerText = `${wasteFactor}%`;

        // Reveal the visualization card panel inside the workspace view
        document.getElementById("upload-stats-panel").style.display = "block";
    }
});

2. The Embedded Vector Search Implementation Engine (src/reaper/engine/search_engine.py)
This script initializes your search utility. It reads your local markdown documentation tree, splits the content into digestible sections, generates technical vector embeddings using gemini-3-flash, and runs a cosine-similarity matrix match to find the most relevant document sections instantly.

Python
# src/reaper/engine/search_engine.py
import os
import glob
from google import genai
from google.genai import types

class DocSearchEngine:
    def __init__(self):
        api_key = os.getenv("GEMINI_API_KEY")
        if not api_key:
            raise ValueError("CRITICAL: GEMINI_API_KEY environment variable is unconfigured.")
        self.client = genai.Client(api_key=api_key)
        self.embedding_model = "text-embedding-004" # Standard structural text embedding generation model
        self.docs_index = []

    def load_and_index_docs(self, docs_dir: str = "docs"):
        """Reads and indexes all markdown files from the target repository documentation tree."""
        self.docs_index = []
        search_path = os.path.join(docs_dir, "**/*.md")
        
        for file_path in glob.glob(search_path, recursive=True):
            with open(file_path, "r", encoding="utf-8") as f:
                content = f.read()
                
            # Naive chunking strategy split by structural section headers
            chunks = content.split("\n## ")
            for idx, chunk in enumerate(chunks):
                if not chunk.strip():
                    continue
                
                clean_chunk = chunk if idx == 0 else f"## {chunk}"
                # Generate embedding vectors for this specific document block
                response = self.client.models.embed_content(
                    model=self.embedding_model,
                    contents=clean_chunk
                )
                
                self.docs_index.append({
                    "file_name": os.path.basename(file_path),
                    "text": clean_chunk,
                    "vector": response.embeddings[0].values
                })

    def query_docs(self, user_query: str, top_k: int = 3) -> list:
        """Executes a contextual vector similarity search against the indexed document arrays."""
        if not self.docs_index:
            return []

        # Generate embedding vector for the inbound search query
        query_response = self.client.models.embed_content(
            model=self.embedding_model,
            contents=user_query
        )
        query_vector = query_response.embeddings[0].values

        # Perform a mathematical Cosine Similarity cross-reference calculation
        scored_results = []
        for doc in self.docs_index:
            dot_product = sum(q * d for q, d in zip(query_vector, doc["vector"]))
            q_norm = sum(q * q for q in query_vector) ** 0.5
            d_norm = sum(d * d for d in doc["vector"]) ** 0.5
            
            similarity = dot_product / (q_norm * d_norm) if (q_norm * d_norm) > 0 else 0
            scored_results.append((similarity, doc))

        # Sort by highest match percentage and return top matches
        scored_results.sort(key=lambda x: x[0], reverse=True)
        return [
            {
                "file": item[1]["file_name"],
                "content": item[1]["text"][:300] + "...", # Snip preview string length
                "confidence_score": f"{item[0] * 100:.2f}%"
            }
            for item in scored_results[:top_k]
        ]

from flask import Blueprint, request, jsonify
from reaper.engine.search_engine import DocSearchEngine

search_bp = Blueprint('search_api', __name__)

# Pre-load and index the markdown files into memory on server initialization

    search_engine.load_and_index_docs(docs_dir="docs")
except Exception as e:

@search_bp.route('/api/v1/docs/search', methods=['POST'])
def handle_docs_search():
    payload = request.get_json() or {}
    query_string = payload.get("query", "").strip()

    if not query_string:
        return jsonify({"status": "error", "message": "Search query string parameter cannot be blank."}), 400

    try:

            "results": search_hits
        }), 200
